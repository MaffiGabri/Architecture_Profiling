# ==============================================================================
# Architecture Profiling Companion - Windows CMake Configuration
# Modern C++20 Desktop Application (Dear ImGui + DirectX 11 / Win32)
# ==============================================================================
cmake_minimum_required(VERSION 3.20)

project(ArchitectureProfilingWindows
    VERSION 1.0.0
    DESCRIPTION "Architecture Profiling Desktop Companion for Windows"
    LANGUAGES CXX C
)

# ------------------------------------------------------------------------------
# 1. C++20 Standard Configuration
# ------------------------------------------------------------------------------
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Output directory layout
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

# ------------------------------------------------------------------------------
# 2. Compiler Warning and Conformance Flags
# ------------------------------------------------------------------------------
if(MSVC)
    add_compile_options(
        /W4                 # Baseline warning level 4
        /permissive-        # Standard C++ conformance mode
        /utf-8              # UTF-8 source and execution character sets
        /Zc:__cplusplus     # Report standard C++ version in __cplusplus macro
        /EHsc               # Standard C++ stack unwinding exception handling
        /MP                 # Multi-processor compilation
        /DNOMINMAX          # Prevent Windows.h from defining min/max macros
        /DWIN32_LEAN_AND_MEAN
    )
    add_compile_options("$<$<CONFIG:Release>:/O2;/GL>")
    add_link_options("$<$<CONFIG:Release>:/LTCG>")
else()
    # MinGW-w64 / GCC / Clang
    add_compile_options(
        -Wall
        -Wextra
        -Wpedantic
        -Wno-unused-parameter
        -DNOMINMAX
        -DWIN32_LEAN_AND_MEAN
    )
    add_compile_options("$<$<CONFIG:Release>:-O3>")
endif()

# ------------------------------------------------------------------------------
# 3. Pure C++20 Domain Library (Milestone M3)
# ------------------------------------------------------------------------------
file(GLOB_RECURSE DOMAIN_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/src/domain/*.cpp"
)
file(GLOB DOMAIN_HEADERS
    "${CMAKE_CURRENT_SOURCE_DIR}/include/architecture/*.hpp"
)
file(GLOB DATA_MODEL_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/src/data/profile_repository.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/data/localization.cpp"
)

# If source files are not yet created, use a clean stub
if(NOT DOMAIN_SOURCES)
    set(DOMAIN_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/domain_stub.cpp")
    if(NOT EXISTS "${DOMAIN_SOURCES}")
        file(WRITE "${DOMAIN_SOURCES}" "// Domain stub generated by CMake\nnamespace arch {\n    const char* version() { return \"1.0.0\"; }\n}\n")
    endif()
endif()

add_library(architecture_domain STATIC
    ${DOMAIN_SOURCES}
    ${DOMAIN_HEADERS}
    ${DATA_MODEL_SOURCES}
)
target_include_directories(architecture_domain PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/src"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/stb"
)

target_link_libraries(architecture_domain PUBLIC
    shell32
    ole32
    shlwapi
)

# ------------------------------------------------------------------------------
# 4. Windows Desktop Application Executable
# ------------------------------------------------------------------------------
file(GLOB IMGUI_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui/imgui.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui/imgui_demo.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui/imgui_draw.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui/imgui_tables.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui/imgui_widgets.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui/backends/imgui_impl_win32.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui/backends/imgui_impl_dx11.cpp"
)

file(GLOB_RECURSE APP_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/platform/*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/ui/*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/data/texture_manager.cpp"
)

add_executable(architecture_windows WIN32 ${APP_SOURCES} ${IMGUI_SOURCES})

target_include_directories(architecture_windows PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/src"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui/backends"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/stb"
)

# Link Domain, DirectX 11, and Win32 Subsystems
target_link_libraries(architecture_windows PRIVATE
    architecture_domain
    d3d11
    dxgi
    d3dcompiler
    dwmapi
    gdi32
    user32
    shell32
    ole32
    shlwapi
)

if(NOT MSVC)
    target_link_options(architecture_windows PRIVATE -mwindows)
endif()

# ------------------------------------------------------------------------------
# 5. Post-Build Asset Deployment
# Synchronizes mock images and styles.json from ../shared into build output
# ------------------------------------------------------------------------------
set(SHARED_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../shared")
if(EXISTS "${SHARED_ROOT}/images" AND EXISTS "${SHARED_ROOT}/data")
    add_custom_command(TARGET architecture_windows POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:architecture_windows>/assets/images"
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:architecture_windows>/assets/data"
        COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${SHARED_ROOT}/images"
            "$<TARGET_FILE_DIR:architecture_windows>/assets/images"
        COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${SHARED_ROOT}/data"
            "$<TARGET_FILE_DIR:architecture_windows>/assets/data"
        COMMENT "Deploying shared assets to build directory..."
    )
endif()

# ------------------------------------------------------------------------------
# 6. Unit Testing and CTest Integration
# ------------------------------------------------------------------------------
enable_testing()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
    add_subdirectory(tests)
endif()
